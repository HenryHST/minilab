# Example secrets for BookStack (NOT applied by Argo — create via SecretSpec / Ansible).
# Namespace: bookstack
#
# Generate APP_KEY:
#   docker run --rm --entrypoint /bin/bash lscr.io/linuxserver/bookstack:version-v26.05.5 appkey
#
# Authentik: OAuth2/OIDC Provider slug `bookstack`
#   Redirect URI: https://book.stadthagen.dev/oidc/callback
#   Groups (optional, matched to BookStack role display names): Admin, Editor
#
# SMTP: password comes from Ansible (--tags secrets) via SecretSpec AUTHENTIK_EMAIL_PASSWORD
#   (same mailbox as Authentik: auto@henrystadthagen.de @ mail.henrystadthagen.de:465).
#   Do not apply this example Secret — Argo must not own bookstack-smtp.
---
apiVersion: v1
kind: Secret
metadata:
  name: bookstack-app
  namespace: bookstack
type: Opaque
stringData:
  APP_KEY: "base64:REPLACE_WITH_APPKEY_OUTPUT"
---
apiVersion: v1
kind: Secret
metadata:
  name: bookstack-db
  namespace: bookstack
type: Opaque
stringData:
  MARIADB_PASSWORD: "REPLACE_DB_PASSWORD"
  MARIADB_ROOT_PASSWORD: "REPLACE_ROOT_PASSWORD"
---
apiVersion: v1
kind: Secret
metadata:
  name: bookstack-oauth
  namespace: bookstack
type: Opaque
stringData:
  client-secret: "REPLACE_AUTHENTIK_CLIENT_SECRET"
---
apiVersion: v1
kind: Secret
metadata:
  name: bookstack-smtp
  namespace: bookstack
type: Opaque
stringData:
  # Placeholder only — real value: secretspec AUTHENTIK_EMAIL_PASSWORD → ansible --tags secrets
  password: "REPLACE_VIA_SECRETSPEC_AUTHENTIK_EMAIL_PASSWORD"
