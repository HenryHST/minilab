# Examples only — real Secrets come from Ansible SecretSpec (--tags secrets).
# Do not commit real values. Do not place these files under the Argo path as *.yaml.
---
# paperless-db
apiVersion: v1
kind: Secret
metadata:
  name: paperless-db
  namespace: paperless
type: Opaque
stringData:
  POSTGRES_PASSWORD: "CHANGE_ME"
---
# paperless-app
apiVersion: v1
kind: Secret
metadata:
  name: paperless-app
  namespace: paperless
type: Opaque
stringData:
  PAPERLESS_SECRET_KEY: "CHANGE_ME_LONG_RANDOM"
  PAPERLESS_ADMIN_USER: "admin"
  PAPERLESS_ADMIN_PASSWORD: "CHANGE_ME"
---
# paperless-oauth — SOCIALACCOUNT JSON embeds Authentik client secret
apiVersion: v1
kind: Secret
metadata:
  name: paperless-oauth
  namespace: paperless
type: Opaque
stringData:
  PAPERLESS_SOCIALACCOUNT_PROVIDERS: >
    {"openid_connect":{"APPS":[{"provider_id":"authentik","name":"Authentik","client_id":"paperless","secret":"CHANGE_ME_OAUTH_SECRET","settings":{"server_url":"https://idp.stadthagen.dev/application/o/paperless/"}}],"OAUTH_PKCE_ENABLED":"True"}}
---
